cmake_minimum_required(VERSION 3.25)
project(ashp_hip_probe LANGUAGES CXX HIP)
find_package(hip REQUIRED CONFIG)
message(STATUS "hip_DIR=${hip_DIR} hip_VERSION=${hip_VERSION}")
message(STATUS "CMAKE_HIP_COMPILER=${CMAKE_HIP_COMPILER} version=${CMAKE_HIP_COMPILER_VERSION}")
add_library(saxpy SHARED src/saxpy.hip)
target_link_libraries(saxpy PUBLIC hip::host)
set_target_properties(saxpy PROPERTIES HIP_ARCHITECTURES gfx1151 INSTALL_RPATH "/opt/rocm/lib")
add_executable(saxpy_probe src/main.cpp)
target_link_libraries(saxpy_probe PRIVATE saxpy)
set_target_properties(saxpy_probe PROPERTIES INSTALL_RPATH "$ORIGIN/../lib:/opt/rocm/lib")
install(TARGETS saxpy saxpy_probe)
